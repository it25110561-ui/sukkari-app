        let state = { user: null, patients: [], selDocPatId: null, selChatPatId: null };

        function showLoginStatus(msg, type) {
            const box = document.getElementById('loginStatusBox');
            box.classList.remove('hidden', 'bg-blue-50', 'text-blue-700', 'border-blue-200', 'bg-red-50', 'text-red-700', 'border-red-200', 'bg-emerald-50', 'text-emerald-700', 'border-emerald-200');
            if (type === 'loading') {
                box.classList.add('bg-blue-50', 'text-blue-700', 'border-blue-200');
            } else if (type === 'error') {
                box.classList.add('bg-red-50', 'text-red-700', 'border-red-200');
            } else {
                box.classList.add('bg-emerald-50', 'text-emerald-700', 'border-emerald-200');
            }
            box.innerText = msg;
        }

        function togglePasswordVisibility(inputId, iconId) {
            const input = document.getElementById(inputId);
            const icon = document.getElementById(iconId);
            if (input.type === 'password') {
                input.type = 'text';
                icon.className = 'fa-solid fa-eye-slash text-emerald-600';
            } else {
                input.type = 'password';
                icon.className = 'fa-solid fa-eye';
            }
        }

        function initTheme() {
            if (localStorage.getItem('theme') === 'dark') {
                document.documentElement.classList.add('dark');
                document.getElementById('themeIcon').className = 'fa-solid fa-sun';
            }
        }

        function toggleTheme() {
            const html = document.documentElement;
            const icon = document.getElementById('themeIcon');
            if (html.classList.contains('dark')) {
                html.classList.remove('dark');
                localStorage.setItem('theme', 'light');
                icon.className = 'fa-solid fa-moon';
            } else {
                html.classList.add('dark');
                localStorage.setItem('theme', 'dark');
                icon.className = 'fa-solid fa-sun';
            }
            if (state.user) {
                state.user.role === 'doctor' ? selectDocPat(state.selDocPatId) : loadPatientData();
            }
        }

        async function initSession() {
            try {
                const r = await fetch('/api/me');
                const d = await r.json();
                if(d.logged_in){
                    state.user = d.user;
                    document.getElementById('authScreen').classList.add('hidden');
                    applyUI();
                }
            } catch(e) {}
        }

        async function performLogin(username, password) {
            const btn = document.getElementById('btnLoginSubmit');
            const origText = btn.innerText;
            btn.disabled = true;
            btn.innerText = 'لحظة دا يفوت...';
            showLoginStatus('⏳ جاري الاتصال بالسيرفر وتسجيل الدخول...', 'loading');

            try {
                const r = await fetch('/api/login', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({username: username, password: password})
                });
                const text = await r.text();
                let d;
                try {
                    d = JSON.parse(text);
                } catch (parseErr) {
                    showLoginStatus('❌ خطأ بالسيرفر (' + r.status + '): تأكد من تحديث ملف app.py', 'error');
                    return;
                }
                if (d.success) {
                    showLoginStatus('✅ تم الدخول بنجاح! جاري فتح حسابك...', 'success');
                    state.user = d.user;
                    setTimeout(() => {
                        document.getElementById('authScreen').classList.add('hidden');
                        applyUI();
                    }, 300);
                } else {
                    showLoginStatus('❌ ' + (d.message || 'فشل تسجيل الدخول'), 'error');
                }
            } catch (err) {
                showLoginStatus('⚠️ ماكو اتصال بالسيرفر: ' + err.message, 'error');
            } finally {
                btn.disabled = false;
                btn.innerText = origText;
            }
        }

        async function handleLogin() {
            const u = document.getElementById('loginUsername').value;
            const p = document.getElementById('loginPassword').value;
            if (!u || !p) {
                showLoginStatus('⚠️ اكتب اليوزر والرمز السري بالبداية', 'error');
                return;
            }
            await performLogin(u, p);
        }

            document.getElementById('loginUsername').value = r;

            const originalText = btn.innerHTML;

            btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> لحظة دا نحفظ...';

            const payload = {
                name: document.getElementById('profName').value,
                age: document.getElementById('profAge').value,
                weight: document.getElementById('profWeight').value,
                height: document.getElementById('profHeight').value,
                emergencyPhone: document.getElementById('profPhone').value,
                newPassword: document.getElementById('profNewPass').value
            };

            try {
                const r = await fetch('/api/patient/update-profile', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify(payload)
                });
                const d = await r.json();
                if (d.success) {
                    state.user = d.user;
                    document.getElementById('navUserName').innerText = state.user.name;
                    document.getElementById('profNewPass').value = '';
                    loadPatientData();

                } else {
                    alert('❌ صار خطأ: ' + (d.message || 'ما كدرنا نحدث البيانات'));
                }
            } catch (err) {
                alert('❌ أكو مشكلة بالاتصال بالسيرفر.');
            } finally {
                btn.disabled = false;
                btn.innerHTML = originalText;
            }
        }

        async function loadPatientData() {
            const r = await fetch('/api/logs'); const logs = await r.json();
            const u = state.user;
            
            document.getElementById('pLatest').innerText = logs.length ? logs[0].val : '-';
            const avg = logs.length ? Math.round(logs.reduce((s,l)=>s+l.val,0)/logs.length) : 0;
            document.getElementById('pAvg').innerText = avg || '-';
            document.getElementById('pA1c').innerText = avg ? ((avg+46.7)/28.7).toFixed(1)+'%' : '-';
            document.getElementById('pBmi').innerText = u.bmi || '-';
            document.getElementById('pSos').innerText = u.emergencyPhone;

            const historyBody = document.getElementById('patientHistoryTable');
            if (logs.length === 0) {
                historyBody.innerHTML = '<tr><td colspan="5" class="p-4 text-center text-slate-400">ماكو أي قياس مسجل لحد هسه.. بلّش سجّل أول قياس إلك!</td></tr>';
            } else {
                historyBody.innerHTML = logs.map(l => {
                    let badge = '<span class="px-2 py-0.5 rounded-full bg-emerald-100 text-emerald-700 font-bold text-[10px]">طبيعي وزين</span>';
                    if (l.val < 70) badge = '<span class="px-2 py-0.5 rounded-full bg-amber-100 text-amber-700 font-bold text-[10px]">هبوط!</span>';
                    else if (l.val > 180) badge = '<span class="px-2 py-0.5 rounded-full bg-red-100 text-red-700 font-bold text-[10px]">صاعد!</span>';
                    return `
                        <tr class="hover:bg-slate-50 dark:hover:bg-slate-800/50 transition">
                            <td class="p-3 text-slate-500">${l.date}</td>
                            <td class="p-3 font-black text-slate-800 dark:text-white">${l.val}</td>
                            <td class="p-3">${l.timing}</td>
                            <td class="p-3 text-slate-500">${l.meds}</td>
                            <td class="p-3">${badge}</td>
                        </tr>
                    `;
                }).join('');
            }
            
            const isDark = document.documentElement.classList.contains('dark');
            const gridColor = isDark ? 'rgba(255,255,255,0.08)' : 'rgba(0,0,0,0.06)';
            const textColor = isDark ? '#94a3b8' : '#64748b';

            if(pc) pc.destroy();
            const reversedLogs = [...logs].reverse();
            pc = new Chart(document.getElementById('pChart'), {
                type:'line',
                data:{
                    labels: reversedLogs.map(l=>l.date.split(' ')[1] || l.date),
                    datasets:[{label:'مستوى السكر', data: reversedLogs.map(l=>l.val), borderColor:'#059669', tension:0.35, fill:true, backgroundColor:'rgba(5,150,105,0.12)'}]
                },
                options:{
                    responsive:true, maintainAspectRatio:false,
                    scales: { x: { ticks: { color: textColor }, grid: { color: gridColor } }, y: { ticks: { color: textColor }, grid: { color: gridColor } } },
                    plugins: { legend: { labels: { color: textColor } } }
                }
            });
        }

        async function addPatient() {
            const payload = {
                name: document.getElementById('nName').value,
                age: document.getElementById('nAge').value,
                weight: document.getElementById('nWeight').value,
                height: document.getElementById('nHeight').value,
                diabetesType: document.getElementById('nDiabetesType').value,
                emergencyPhone: document.getElementById('nPhone').value,
                username: document.getElementById('nUser').value,
                password: document.getElementById('nPass').value
            };
            if (!payload.name || !payload.username || !payload.password) {
                alert('املأ الاسم واليوزر والرمز السري بالبداية');
                return;
            }
            
            const r = await fetch('/api/doctor/create-patient', {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify(payload)});
            const d = await r.json();
            if (!d.success) {
                alert(d.message || 'صار خطأ أثناء إضافة المراجع');
                return;
            }
            document.getElementById('addModal').classList.add('hidden');
            document.getElementById('nName').value = '';
            document.getElementById('nPhone').value = '';
            document.getElementById('nUser').value = '';
            document.getElementById('nPass').value = '';

            state.selDocPatId = d.patient_id;
            await loadDoctorData();
        }
